array

import tinox.core.array;

Utility-Klasse für Operationen auf Integer-Arrays: Suchen, Transformieren, Sortieren und Generieren von Ranges.

Klassen

Arrays

Stellt statische Methoden für Listen von Int64-Werten bereit.

MethodeSignaturBeschreibung
containsIntfn containsInt(arr: List<Int64>, value: Int64) -> BoolPrüft ob ein Wert enthalten ist
indexOfIntfn indexOfInt(arr: List<Int64>, value: Int64) -> Int64Index des ersten Vorkommens, -1 wenn nicht gefunden
findMinfn findMin(arr: List<Int64>) -> Int64Kleinstes Element (Array darf nicht leer sein)
findMaxfn findMax(arr: List<Int64>) -> Int64Größtes Element (Array darf nicht leer sein)
sumfn sum(arr: List<Int64>) -> Int64Summe aller Elemente
averagefn average(arr: List<Int64>) -> Int64Ganzzahliger Durchschnitt
reverseIntfn reverseInt(arr: List<Int64>) -> List<Int64>Neues Array in umgekehrter Reihenfolge
filterPositivefn filterPositive(arr: List<Int64>) -> List<Int64>Nur positive Werte (> 0)
bubbleSortfn bubbleSort(arr: List<Int64>) -> List<Int64>Sortiert aufsteigend via Bubble Sort
isSortedfn isSorted(arr: List<Int64>) -> BoolPrüft ob Array aufsteigend sortiert ist
rangefn range(start: Int64, end: Int64) -> List<Int64>Erzeugt Range [start, end)
fillfn fill(value: Int64, count: Int64) -> List<Int64>Array mit count Kopien von value

Beispiel

import tinox.core.array;

fn main() -> Int64
{
    let nums: List<Int64> = [5, 2, 8, 1, 9, 3];

    // Suchen
    let found: Bool = Arrays::containsInt(nums, 8);   // true
    let idx: Int64  = Arrays::indexOfInt(nums, 8);    // 2

    // Statistik
    let total: Int64 = Arrays::sum(nums);             // 28
    let min: Int64   = Arrays::findMin(nums);          // 1
    let max: Int64   = Arrays::findMax(nums);          // 9

    // Transformieren
    let sorted:   List<Int64> = Arrays::bubbleSort(nums);
    let reversed: List<Int64> = Arrays::reverseInt(nums);
    let positive: List<Int64> = Arrays::filterPositive([-1, 2, -3, 4]);

    // Generieren
    let r: List<Int64> = Arrays::range(0, 5);    // [0,1,2,3,4]
    let f: List<Int64> = Arrays::fill(42, 3);   // [42,42,42]

    return 0;
}
