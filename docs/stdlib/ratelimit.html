ratelimit

import tinox.core.ratelimit;

Rate-Limiting: Sliding-Window RateLimiter und Token-Bucket-Algorithmus für Anfragen-Drosselung.

Klassen

RateLimiter

MethodeSignaturBeschreibung
newfn new(maxRequests: Int64, windowMs: Int64) -> RateLimiterSliding-Window-Limiter erstellen
allowfn allow(limiter: RateLimiter) -> BoolAnfrage prüfen und registrieren
resetfn reset(limiter: RateLimiter) -> UnitAnfrage-Historie zurücksetzen
remainingfn remaining(limiter: RateLimiter) -> Int64Verbleibende Anfragen im Fenster

TokenBucket

MethodeSignaturBeschreibung
newfn new(capacity: Int64, refillRate: Float64) -> TokenBucketBucket mit Kapazität und Nachfüllrate (Token/s)
allowfn allow(bucket: TokenBucket) -> BoolToken verbrauchen wenn verfügbar

Beispiel

import tinox.core.ratelimit;

fn main() -> Int64
{
    // Max. 5 Anfragen pro Sekunde (Sliding Window)
    let limiter: RateLimiter = RateLimiter::new(5, 1000);
    if RateLimiter::allow(limiter) {
        println("Anfrage erlaubt");
    } else {
        println("Rate Limit überschritten");
    }
    println(RateLimiter::remaining(limiter).toString());

    // Token Bucket: 10 Tokens, 2 Token/s Nachfüllung
    let bucket: TokenBucket = TokenBucket::new(10, 2.0);
    println(TokenBucket::allow(bucket).toString()); // "true"
    return 0;
}
