semaphore

import tinox.core.semaphore;

Synchronisationsprimitiven: Semaphore, Mutex und RWLock (Reader-Writer-Lock).

Klassen

Semaphore

MethodeSignaturBeschreibung

releasefnc release(sem: Semaphore) -> UnitZähler inkrementieren

RWLock

MethodeSignaturBeschreibung

readLockfnc readLock(lock: RWLock) -> UnitLesesperre (blockiert bei aktivem Schreiber)
readUnlockfnc readUnlock(lock: RWLock) -> UnitLesesperre freigeben
writeLockfnc writeLock(lock: RWLock) -> UnitExklusivsperre für Schreiber
writeUnlockfnc writeUnlock(lock: RWLock) -> UnitSchreibsperre freigeben

Beispiel

import tinox.core.semaphore;

fn main() -> Int64
{
    let sem: Semaphore = Semaphore::new(3); // max 3 gleichzeitig
    Semaphore::acquire(sem);
    // kritischer Abschnitt
    Semaphore::release(sem);

    let mtx: Mutex = Mutex::new();
    if Mutex::tryLock(mtx) {
        // exklusiver Zugriff
        Mutex::unlock(mtx);
    }
    return 0;
}
