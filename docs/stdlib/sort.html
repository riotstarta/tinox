Sortieralgorithmen für Integer-Listen: QuickSort, MergeSort und InsertionSort.

Klassen

Sort

MethodeSignaturKomplexität
quickSortfn quickSort(arr: List<Int64>) -> List<Int64>O(n log n) ø
mergeSortfn mergeSort(arr: List<Int64>) -> List<Int64>O(n log n)
insertionSortfn insertionSort(arr: List<Int64>) -> List<Int64>O(n²)

Beispiel

import tinox.core.sort;

fn main() -> Int64
{
    let nums: List<Int64> = [5, 2, 8, 1, 9, 3];

    let sorted: List<Int64> = Sort::quickSort(nums);
    // sorted = [1, 2, 3, 5, 8, 9]

    let merged: List<Int64> = Sort::mergeSort(nums);
    // merged = [1, 2, 3, 5, 8, 9]

    let small: List<Int64> = [3, 1, 2];
    let ins: List<Int64> = Sort::insertionSort(small);
    // ins = [1, 2, 3]
    return 0;
}
