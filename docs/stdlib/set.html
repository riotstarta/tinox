set

import tinox.core.set;

Generische Menge (Set) ohne Duplikate: add, remove, contains und Mengenoperationen (union, intersection, difference).

Klassen

Set<T>

MethodeSignaturBeschreibung
newfnc new() -> Set<T>Leere Menge erstellen
addfnc add(set: Set<T>, item: T) -> UnitElement hinzufügen (Duplikate ignoriert)
removefnc remove(set: Set<T>, item: T) -> BoolElement entfernen, true wenn gefunden
containsfnc contains(set: Set<T>, item: T) -> BoolEnthält Menge dieses Element?
sizefnc size(set: Set<T>) -> Int64Anzahl Elemente
isEmptyfnc isEmpty(set: Set<T>) -> BoolIst die Menge leer?
unionfnc union(a: Set<T>, b: Set<T>) -> Set<T>Vereinigung beider Mengen
intersectionfnc intersection(a: Set<T>, b: Set<T>) -> Set<T>Schnittmenge
differencefnc difference(a: Set<T>, b: Set<T>) -> Set<T>Differenzmenge (a ohne b)
isSubsetOffnc isSubsetOf(a: Set<T>, b: Set<T>) -> BoolIst a Teilmenge von b?

Beispiel

import tinox.core.set;

fn main() -> Int64
{
    let a: Set<Int64> = Set::new();
    Set::add(a, 1); Set::add(a, 2); Set::add(a, 3);
    Set::add(a, 2); // Duplikat – ignoriert

    let b: Set<Int64> = Set::new();
    Set::add(b, 2); Set::add(b, 3); Set::add(b, 4);

    let u: Set<Int64> = Set::union(a, b);        // {1,2,3,4}
    let i: Set<Int64> = Set::intersection(a, b); // {2,3}
    let d: Set<Int64> = Set::difference(a, b);    // {1}
    println(Set::size(u).toString()); // "4"
    return 0;
}
