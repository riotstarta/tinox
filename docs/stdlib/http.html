http

import tinox.core.http;

HTTP-Client für GET, POST, PUT, DELETE und PATCH-Anfragen mit Header-Verwaltung.

Klassen

Http

MethodeSignaturBeschreibung
getfnc get(url: String) -> HttpResponseGET-Anfrage senden
postfnc post(url: String, body: String) -> HttpResponsePOST-Anfrage mit Body senden
putfnc put(url: String, body: String) -> HttpResponsePUT-Anfrage senden
deletefnc delete(url: String) -> HttpResponseDELETE-Anfrage senden
patchfnc patch(url: String, body: String) -> HttpResponsePATCH-Anfrage senden

HttpResponse

MethodeSignaturBeschreibung
statusCodefnc statusCode() -> Int64HTTP-Statuscode (z.B. 200, 404)
bodyfnc body() -> StringAntwort-Body als String
headerfnc header(name: String) -> StringEinzelnen Antwort-Header lesen

Beispiel

import tinox.core.http;

fn main() -> Int64
{
    Http::setHeader("Accept", "application/json");

    let res: HttpResponse = Http::get("https://api.example.com/users");
    println(res.statusCode().toString()); // "200"
    println(res.body());

    let created: HttpResponse = Http::post(
        "https://api.example.com/users",
        "{\"name\":\"Alice\"}"
    );
    println(created.statusCode().toString()); // "201"
    return 0;
}
