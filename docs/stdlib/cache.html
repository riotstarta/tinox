cache

import tinox.core.cache;

Generischer LRU-Cache mit konfigurierbarer Maximalgröße. Älteste Einträge werden automatisch verdrängt, wenn der Cache voll ist.

Klassen

Cache<K, V>

Basis-Cache mit LRU-Eviction-Strategie.

MethodeSignaturBeschreibung

Beispiel

import tinox.core.cache;

fn main() -> Int64
{
    // LRU-Cache mit max. 3 Einträgen
    let cache: LruCache<String, Int64> = LruCache::new(3);

    LruCache::set(cache, "a", 1);
    LruCache::set(cache, "b", 2);
    LruCache::set(cache, "c", 3);

    let val: Option<Int64> = LruCache::get(cache, "a");
    if val.isSome()
    {
        println(val.unwrap().toString()); // "1"
    }

    // "d" setzen verdrängt den ältesten Eintrag (b)
    LruCache::set(cache, "d", 4);
    return 0;
}
