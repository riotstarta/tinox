encoding

import tinox.core.encoding;

Text-Encoding-Utilities: Zeichencode-Konvertierung, UTF-8-Byte-Konvertierung und HTML-Entity-Kodierung/-Dekodierung.

Klassen

Encoding

MethodeSignaturBeschreibung
toCharCodefn toCharCode(s: String) -> Int64Erstes Zeichen als Unicode-Codepoint
fromCharCodefn fromCharCode(code: Int64) -> StringCodepoint zu Zeichen
toUtf8fn toUtf8(s: String) -> List<Int64>String in UTF-8 Byte-Array
fromUtf8fn fromUtf8(codes: List<Int64>) -> StringByte-Array zurück zu String
htmlEncodefn htmlEncode(s: String) -> StringHTML-Entities kodieren (&, <, >, ", ')
htmlDecodefn htmlDecode(s: String) -> StringHTML-Entities dekodieren

Beispiel

import tinox.core.encoding;

fn main() -> Int64
{
    // HTML-Encoding für sichere Ausgabe
    let userInput: String = "<script>alert('XSS')</script>";
    let safe: String = Encoding::htmlEncode(userInput);
    println(safe); // "&lt;script&gt;alert(&#39;XSS&#39;)..."

    // Zeichencode-Konvertierung
    let code: Int64  = Encoding::toCharCode("A");  // 65
    let ch:   String = Encoding::fromCharCode(65); // "A"

    // UTF-8 Bytes
    let bytes: List<Int64> = Encoding::toUtf8("Hi"); // [72, 105]
    return 0;
}
