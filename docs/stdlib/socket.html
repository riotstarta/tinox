socket

import tinox.core.socket;

TCP- und UDP-Socket-Abstraktion mit connect, bind, listen, accept, send und receive.

Klassen

Socket

MethodeSignaturBeschreibung
createTcpfnc createTcp() -> SocketTCP-Socket erstellen
createUdpfnc createUdp() -> SocketUDP-Socket erstellen
connectfnc connect(socket: Socket, host: String, port: Int64) -> BoolZu Server verbinden
bindfnc bind(socket: Socket, port: Int64) -> BoolSocket an Port binden
listenfnc listen(socket: Socket) -> BoolAuf eingehende Verbindungen warten
acceptfnc accept(socket: Socket) -> SocketNeue Client-Verbindung annehmen
sendfnc send(socket: Socket, data: String) -> Int64Daten senden, gibt gesendete Bytes zurück
receivefnc receive(socket: Socket, size: Int64) -> StringBis zu size Bytes empfangen
closefnc close(socket: Socket) -> UnitVerbindung schließen

Beispiel

import tinox.core.socket;

fn main() -> Int64
{
    // TCP-Client
    let client: Socket = Socket::createTcp();
    if Socket::connect(client, "localhost", 8080) {
        Socket::send(client, "Hallo Server!");
        let response: String = Socket::receive(client, 1024);
        println(response);
        Socket::close(client);
    }
    return 0;
}
