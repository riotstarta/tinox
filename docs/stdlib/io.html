import tinox.core.io;

Beispiel

import tinox.core.io;

fn main() -> Int64
{
    let buf: Buffer = Buffer::new();
    buf.appendLine("Zeile 1");
    buf.appendLine("Zeile 2");
    buf.appendInt(42);
    println(buf.toString());

    // Pfad-Utilities
    let ext: String = Paths::getExtension("main.tnx");
    println(ext); // "tnx"

    let dir: String = Paths::getDirectory("/src/main.tnx");
    println(dir); // "/src"
    return 0;
}
