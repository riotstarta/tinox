Beispiel

import tinox.core.string;

fn main() -> Int64
{
    let s: String = "  Hallo Tinox!  ";
    println(Strings::trim(s));                     // "Hallo Tinox!"
    println(Strings::toUpperCase("abc"));           // "ABC"
    println(Strings::repeat("ab", 3));              // "ababab"
    println(Strings::reverse("Tinox"));             // "xoniT"

    let parts: List<String> = Strings::split("a,b,c", ",");
    let joined: String = Strings::join(parts, " | ");
    println(joined); // "a | b | c"
    return 0;
}
