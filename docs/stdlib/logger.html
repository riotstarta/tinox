logger

import tinox.core.logger;

Konfigurierbares Logging mit Level-Filter (Debug, Info, Warn, Error) und Timestamp-Ausgabe.

Klassen

Logger

MethodeSignaturBeschreibung
newfn new(name: String) -> LoggerLogger mit Name erstellen (Standard-Level: Info)
debugfn debug(logger: Logger, message: String) -> NothingDebug-Nachricht (nur wenn Level ≤ Debug)
infofn info(logger: Logger, message: String) -> NothingInfo-Nachricht
warnfn warn(logger: Logger, message: String) -> NothingWarn-Nachricht
errorfn error(logger: Logger, message: String) -> NothingError-Nachricht
setLevelfn setLevel(logger: Logger, level: LogLevel) -> NothingMindest-Log-Level setzen

LogLevel

KonstanteWertBeschreibung
LogLevel::Debug()0Alle Nachrichten ausgeben
LogLevel::Info()1Info, Warn, Error ausgeben
LogLevel::Warn()2Nur Warn und Error ausgeben
LogLevel::Error()3Nur Error ausgeben

Beispiel

import tinox.core.logger;

fn main() -> Int64
{
    let log: Logger = Logger::new("app");
    Logger::setLevel(log, LogLevel::Debug());

    Logger::debug(log, "Starte Initialisierung");
    Logger::info(log, "Server bereit");
    Logger::warn(log, "Speicher fast voll");
    Logger::error(log, "Verbindung fehlgeschlagen");
    // Ausgabe: [timestamp] [DEBUG] [app] Starte Initialisierung
    return 0;
}
