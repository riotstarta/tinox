complex

import tinox.core.complex;

Komplexe Zahlen mit Arithmetik, polaren Koordinaten, trigonometrischen und transzendenten Funktionen.

Beispiel

import tinox.core.complex;

fn main() -> Int64
{
    let a: Complex = Complex::new(3.0, 4.0);  // 3 + 4i
    let b: Complex = Complex::new(1.0, 2.0);  // 1 + 2i

    let sum:  Complex = Complex::add(a, b);       // 4 + 6i
    let prod: Complex = Complex::multiply(a, b); // -5 + 10i
    let mag:  Float64 = Complex::magnitude(a);   // 5.0
    let conj: Complex = Complex::conjugate(a);  // 3 - 4i

    println(Complex::toString(prod));
    return 0;
}
