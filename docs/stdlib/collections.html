collections

import tinox.core.collections;

Datenstruktur-Implementierungen: Stack (LIFO), Queue (FIFO) und Pair. Factory-Klasse Collections für einfache Erzeugung.

Klassen

Stack

LIFO-Stack für Integer-Werte. Wirft Fehler bei Pop/Peek auf leerem Stack.

MethodeSignaturBeschreibung
newfnc new() -> StackLeeren Stack erstellen

peekfn peek() -> Int64Oberstes Element ansehen (ohne Entfernen)

Beispiel

import tinox.core.collections;

fn main() -> Int64
{
    // Stack als Klammerprüfer
    let stack: Stack = Stack::new();
    stack.push(40);  // '('
    stack.push(91);  // '['
    println(stack.pop().toString()); // 91

    // Queue für Aufgabenverarbeitung
    let q: Queue = Queue::new();
    q.enqueue(1);
    q.enqueue(2);
    q.enqueue(3);
    while q.isEmpty() == false
    {
        println(q.dequeue().toString()); // 1, 2, 3
    }

    // Pair für Min/Max-Rückgabe
    let p: Pair = Pair::new(10, 99);
    println(p.getFirst().toString());  // 10
    return 0;
}
