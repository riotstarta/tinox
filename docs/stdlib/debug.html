debug

import tinox.core.debug;

Debug-Utilities: Assertions, Panic, Stack-Trace, Typ-Inspektion und Garbage-Collector-Steuerung.

Klassen

Debug

MethodeSignaturBeschreibung
assertfn assert(condition: Bool, message: String) -> NothingWirft Fehler wenn Bedingung false ist
panicfn panic(message: String) -> NothingUnmittelbarer Programmabbruch mit Nachricht
printStackTracefn printStackTrace() -> NothingAktuellen Call-Stack ausgeben
typeName<T>fn typeName<T>() -> StringName des Typs T als String
memoryUsagefn memoryUsage() -> Int64Aktuellen Speicherverbrauch in Bytes
gcCollectfn gcCollect() -> NothingGarbage Collection manuell auslösen

Beispiel

import tinox.core.debug;

fn divide(a: Int64, b: Int64) -> Int64
{
    Debug::assert(b != 0, "Division durch Null");
    return a / b;
}

fn main() -> Int64
{
    let mem: Int64 = Debug::memoryUsage();
    println("Speicher: " + mem.toString() + " Bytes");

    let result: Int64 = divide(10, 2); // OK
    // divide(10, 0); // wirft: "Assertion failed: Division durch Null"

    Debug::gcCollect();
    return 0;
}
