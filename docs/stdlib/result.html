result

import tinox.core.result;

Result-Typ für Fehlerbehandlung: ok/err Varianten mit sicherem Unwrapping und Transformationen.

Klassen

Result

MethodeSignaturBeschreibung
okfnc ok(value: Int64) -> ResultErfolgreichen Wert einwickeln
errfnc err(error: String) -> ResultFehlerwert mit Nachricht erstellen
isOkfnc isOk() -> BoolIst das Ergebnis erfolgreich?
isErrfnc isErr() -> BoolIst ein Fehler aufgetreten?
unwrapfnc unwrap() -> Int64Wert extrahieren (throws bei Err)
unwrapOrfnc unwrapOr(defaultValue: Int64) -> Int64Wert oder Fallback zurückgeben
mapfnc map(transform: fnc(Int64) -> Int64) -> ResultWert transformieren (Err bleibt Err)

Beispiel

import tinox.core.result;

fn divide(a: Int64, b: Int64) -> Result
{
    if b == 0 { return Result::err("Division durch Null"); }
    return Result::ok(a / b);
}

fn main() -> Int64
{
    let r1: Result = divide(10, 2);
    println(r1.unwrap().toString()); // "5"

    let r2: Result = divide(10, 0);
    println(r2.unwrapOr(-1).toString()); // "-1"
    println(r2.isErr().toString());       // "true"
    return 0;
}
