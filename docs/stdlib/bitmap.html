bitmap

import tinox.core.bitmap;

2D-Pixel-Bitmap mit Zeichenoperationen: Pixel setzen/lesen, Rechtecke füllen, Linien und Kreise zeichnen (Bresenham-Algorithmus).

Klassen

Bitmap

Felder: width: Int64, height: Int64, pixels: List<Int64>

MethodeSignaturBeschreibung
createfn create(width, height: Int64, color: Int64) -> BitmapNeues Bitmap mit Hintergrundfarbe erstellen
getPixelfn getPixel(bitmap, x, y: Int64) -> Int64Pixelfarbe auslesen (0 bei Out-of-Bounds)
setPixelfn setPixel(bitmap, x, y: Int64, color: Int64) -> NothingPixelfarbe setzen
fillRectfn fillRect(bitmap, x, y, w, h: Int64, color: Int64) -> NothingRechteck füllen
drawLinefn drawLine(bitmap, x0, y0, x1, y1: Int64, color: Int64) -> NothingLinie zeichnen (Bresenham)
drawCirclefn drawCircle(bitmap, cx, cy, radius: Int64, color: Int64) -> NothingKreis zeichnen (Midpoint)

Beispiel

import tinox.core.bitmap;

fn main() -> Int64
{
    // 100x100 Bitmap, schwarzer Hintergrund (0x000000)
    let bmp: Bitmap = Bitmap::create(100, 100, 0);

    // Weißes Rechteck (0xFFFFFF)
    Bitmap::fillRect(bmp, 10, 10, 80, 80, 16777215);

    // Rote Linie (0xFF0000)
    Bitmap::drawLine(bmp, 0, 0, 99, 99, 16711680);

    // Blauer Kreis (0x0000FF)
    Bitmap::drawCircle(bmp, 50, 50, 30, 255);

    let color: Int64 = Bitmap::getPixel(bmp, 50, 50);
    return 0;
}
