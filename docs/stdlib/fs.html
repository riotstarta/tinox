fs

import tinox.core.fs;

Dateisystem-Operationen: Lesen, Schreiben, Kopieren, Verschieben, Löschen von Dateien und Verzeichnissen.

Klassen

Fs

MethodeSignaturBeschreibung
readFilefnc readFile(path: String) -> StringDatei vollständig als String lesen
writeFilefnc writeFile(path: String, content: String) -> UnitString in Datei schreiben (überschreibt)
appendFilefnc appendFile(path: String, content: String) -> UnitInhalt an bestehende Datei anhängen
deleteFilefnc deleteFile(path: String) -> UnitDatei löschen
copyFilefnc copyFile(src: String, dest: String) -> UnitDatei kopieren
moveFilefnc moveFile(src: String, dest: String) -> UnitDatei verschieben (kopieren + löschen)
listDirectoryfnc listDirectory(path: String) -> List<String>Verzeichnisinhalt auflisten
createDirectoryfnc createDirectory(path: String) -> UnitVerzeichnis erstellen
deleteDirectoryfnc deleteDirectory(path: String) -> UnitVerzeichnis löschen

Beispiel

import tinox.core.fs;

fn main() -> Int64
{
    // Datei schreiben und lesen
    Fs::writeFile("hello.txt", "Hallo Tinox!");
    let content: String = Fs::readFile("hello.txt");
    println(content); // "Hallo Tinox!"

    // Anhängen
    Fs::appendFile("hello.txt", "\nZeile 2");

    // Verzeichnis anlegen und auflisten
    Fs::createDirectory("output");
    let entries: List<String> = Fs::listDirectory(".");

    // Datei kopieren und löschen
    Fs::copyFile("hello.txt", "output/hello.txt");
    Fs::deleteFile("hello.txt");
    return 0;
}
