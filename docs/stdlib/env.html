env

import tinox.core.env;

Zugriff auf Umgebungsvariablen und das aktuelle Arbeitsverzeichnis des Prozesses.

Klassen

Env

MethodeSignaturBeschreibung
getVarfn getVar(name: String) -> StringUmgebungsvariable lesen
setVarfn setVar(name, value: String) -> NothingUmgebungsvariable setzen
removeVarfn removeVar(name: String) -> NothingUmgebungsvariable entfernen
currentDirfn currentDir() -> StringAktuelles Arbeitsverzeichnis
setCurrentDirfn setCurrentDir(path: String) -> NothingArbeitsverzeichnis wechseln

Beispiel

import tinox.core.env;

fn main() -> Int64
{
    // Konfiguration aus Umgebungsvariablen lesen
    let dbHost: String = Env::getVar("DATABASE_HOST");
    let port:   String = Env::getVar("PORT");

    if dbHost == ""
    {
        Env::setVar("DATABASE_HOST", "localhost");
    }

    let cwd: String = Env::currentDir();
    println("Arbeitsverzeichnis: " + cwd);
    return 0;
}
