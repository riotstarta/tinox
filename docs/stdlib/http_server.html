http_server

import tinox.core.http_server;

Vollständiger HTTP/1.1-Server mit Routing, Middleware-Support, Pfad-Parametern und Query-String-Parsing.

Beispiel

import tinox.core.http_server;

fn main() -> Int64
{
    let server: HttpServer = HttpServer::new(8080);

    server.get("/hello", fnc(ctx: HttpContext) -> Unit {
        ctx.response.text("Hallo Welt!");
    });

    server.get("/users/:id", fnc(ctx: HttpContext) -> Unit {
        let id: String = ctx.request.params["id"];
        ctx.response.status(200).json("{\"id\":" + id + "}");
    });

    server.post("/users", fnc(ctx: HttpContext) -> Unit {
        ctx.response.status(201).json("{\"created\":true}");
    });

    server.listen(); // blockiert
    return 0;
}
