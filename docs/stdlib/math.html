math

import tinox.core.math;

Integer-Mathematik: Konstanten (π, e, τ), abs, min, max, pow, factorial und LCG-Pseudozufallsgenerator.

Beispiel

import tinox.core.math;

fn main() -> Int64
{
    println(Math::pi().toString());         // "3.14159265358979"
    println(Math::pow(2, 10).toString());    // "1024"
    println(Math::factorial(6).toString()); // "720"
    println(Math::abs(-42).toString());      // "42"

    let rnd: Int64 = Math::randomSeed(12345);
    println(rnd.toString());
    return 0;
}
