Beispiel

import tinox.core.rest;

fn main() -> Int64
{
    // Einfacher REST-Client
    let client: RestClient = RestClient::new("https://api.example.com");
    client.bearerAuth("mein-token-123");

    let res: RestResponse = client.get("/users");
    if res.isOk() {
        println(res.text());
    }

    // POST mit JSON-Body
    let body: String = "{\"name\":\"Alice\",\"age\":30}";
    let created: RestResponse = client.post("/users", body);
    println(created.status().toString()); // z.B. "201"

    // Fluenter Builder mit Query-Parametern
    let resp: RestResponse = RequestBuilder::new("https://api.example.com")
        .bearer("mein-token-123")
        .query("page", "1")
        .query("limit", "10")
        .get("/users");

    resp.ensureSuccess();
    let data: JsonValue = resp.json();

    // URL-Building
    let url: String = Url::buildWithParams("https://api.example.com", "/search", {
        "q": "tinox",
        "lang": "de"
    });
    println(url); // "https://api.example.com/search?q=tinox&lang=de"

    return 0;
}
