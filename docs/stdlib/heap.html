MinHeap<T> / MaxHeap<T>

Convenience-Wrapper mit fixen Comparatoren.

Beispiel

import tinox.core.heap;

fn main() -> Int64
{
    let min: MinHeap<Int64> = MinHeap::new();
    MinHeap::push(min, 5);
    MinHeap::push(min, 1);
    MinHeap::push(min, 3);
    println(MinHeap::pop(min).toString()); // "1"
    println(MinHeap::pop(min).toString()); // "3"

    let max: MaxHeap<Int64> = MaxHeap::new();
    MaxHeap::push(max, 5);
    MaxHeap::push(max, 1);
    println(MaxHeap::pop(max).toString()); // "5"
    return 0;
}
