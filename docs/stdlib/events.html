events

import tinox.core.events;

Event-Emitter für das Publish-Subscribe-Pattern. Registriere Handler für benannte Events und emittiere Daten an alle Listener.

Klassen

EventEmitter

MethodeSignaturBeschreibung
newfnc new() -> EventEmitterNeuen EventEmitter erstellen
onfnc on(emitter, event: String, handler: fnc(JsonValue) -> Unit) -> UnitPermanenten Listener registrieren
oncefnc once(emitter, event: String, handler: fnc(JsonValue) -> Unit) -> UnitEinmaligen Listener registrieren
emitfnc emit(emitter, event: String, data: JsonValue) -> UnitEvent an alle Listener senden
removeListenerfnc removeListener(emitter, event: String, handler) -> UnitEinzelnen Listener entfernen
removeAllListenersfnc removeAllListeners(emitter, event: String) -> UnitAlle Listener eines Events entfernen
listenerCountfnc listenerCount(emitter, event: String) -> Int64Anzahl der Listener für ein Event

Beispiel

import tinox.core.events;
import tinox.core.json;

fn main() -> Int64
{
    let emitter: EventEmitter = EventEmitter::new();

    // Listener registrieren
    EventEmitter::on(emitter, "userLogin", fnc(data: JsonValue) -> Unit
    {
        println("User eingeloggt");
    });

    // Einmaliger Listener für Initialisierung
    EventEmitter::once(emitter, "ready", fnc(data: JsonValue) -> Unit
    {
        println("System bereit");
    });

    // Events auslösen
    EventEmitter::emit(emitter, "ready", Json::parse("{}"));
    EventEmitter::emit(emitter, "userLogin", Json::parse("{\"user\":\"alice\"}"));
    return 0;
}
