mathf

import tinox.core.mathf;

Beispiel

import tinox.core.mathf;

fn main() -> Int64
{
    println(Mathf::sqrt(2.0).toString());      // ~"1.41421356..."
    println(Mathf::clamp(1.5, 0.0, 1.0).toString()); // "1.0"
    println(Mathf::lerp(0.0, 10.0, 0.5).toString()); // "5.0"
    println(Mathf::floor(3.9).toString());     // "3"
    println(Mathf::ceil(3.1).toString());      // "4"
    return 0;
}
