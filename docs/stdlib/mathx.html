mathx

import tinox.core.mathx;

Erweiterte Mathematik: GCD, LCM, Primzahlen, Fibonacci, Kombinatorik, modulare Potenz und Ziffernoperationen.

Beispiel

import tinox.core.mathx;

fn main() -> Int64
{
    println(Mathx::gcd(48, 18).toString());     // "6"
    println(Mathx::isPrime(17).toString());      // "true"
    println(Mathx::fibonacci(10).toString());    // "55"
    println(Mathx::combinations(5, 2).toString()); // "10"
    println(Mathx::modPow(2, 10, 1000).toString()); // "24"
    println(Mathx::digitSum(12345).toString());  // "15"
    return 0;
}
