tpl

import tinox.core.tpl;

Template-Engine mit {{variable}}-Substitution, Schleifen-Rendering und vorkompilierten Templates.

Klassen

Template

MethodeSignaturBeschreibung
renderfnc render(template: String, data: Map<String, JsonValue>) -> String{{key}} durch Werte aus data ersetzen
renderLoopfnc renderLoop(template, items, itemVar: String) -> StringTemplate für jedes Element der Liste rendern
compilefnc compile(template: String) -> CompiledTemplateTemplate vorkompilieren (für Wiederverwendung)
executefnc execute(tpl: CompiledTemplate, data: Map<String, JsonValue>) -> StringVorkompiliertes Template ausführen

Beispiel

import tinox.core.tpl;
import tinox.core.json;

fn main() -> Int64
{
    let tmpl: String = "Hallo {{name}}, du bist {{age}} Jahre alt!";
    let data: Map<String, JsonValue> = Map::new();
    data["name"] = Json::parse("\"Alice\"");
    data["age"]  = Json::parse("30");

    let result: String = Template::render(tmpl, data);
    println(result); // "Hallo Alice, du bist 30 Jahre alt!"

    // Vorkompilieren
    let compiled: CompiledTemplate = Template::compile(tmpl);
    println(Template::execute(compiled, data));
    return 0;
}
