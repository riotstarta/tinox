ini

import tinox.core.ini;

Parser und Serialisierer für INI-Konfigurationsdateien mit Sections und Key-Value-Paaren.

Klassen

Ini

MethodeSignaturBeschreibung
parsefnc parse(content: String) -> IniConfigINI-String parsen
stringifyfnc stringify(config: IniConfig) -> StringIniConfig zurück in String serialisieren

IniConfig

MethodeSignaturBeschreibung

getStringfnc getString(config, section, key, default: String) -> StringString-Wert lesen (mit Fallback)
getIntfnc getInt(config, section, key, default: Int64) -> Int64Integer-Wert lesen (mit Fallback)
setStringfnc setString(config, section, key, value: String) -> UnitWert setzen

Beispiel

import tinox.core.ini;

fn main() -> Int64
{
    let raw: String = "[server]\nhost=localhost\nport=8080\n";
    let cfg: IniConfig = Ini::parse(raw);

    let host: String = IniConfig::getString(cfg, "server", "host", "localhost");
    let port: Int64  = IniConfig::getInt(cfg, "server", "port", 80);
    println(host);              // "localhost"
    println(port.toString());   // "8080"

    IniConfig::setString(cfg, "server", "host", "0.0.0.0");
    println(Ini::stringify(cfg));
    return 0;
}
