validation

import tinox.core.validation;

Eingabe-Validierung: E-Mail, URL, numerisch, alpha, alphanumerisch, Telefon, Längen- und Wertebereichs-Prüfungen.

Klassen

Beispiel

import tinox.core.validation;

fn main() -> Int64
{
    println(Validation::isEmail("user@example.com").toString()); // "true"
    println(Validation::isEmail("kein-email").toString());        // "false"
    println(Validation::isNumeric("12345").toString());            // "true"
    println(Validation::isAlphanumeric("abc123").toString());    // "true"
    println(Validation::minLength("passwort", 8).toString());    // "true"
    println(Validation::isInRange(42, 1, 100).toString());        // "true"
    return 0;
}
