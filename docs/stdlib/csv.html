csv

import tinox.core.csv;

CSV-Parser und Serialisierer. Unterstützt gequotete Felder mit Kommas und eingebetteten Anführungszeichen.

Klassen

Csv

MethodeSignaturBeschreibung
parsefnc parse(content: String) -> List<List<String>>CSV-Text in 2D-Liste parsen
parseRowfnc parseRow(line: String) -> List<String>Einzelne Zeile parsen (Quote-aware)
toStringfnc toString(data: List<List<String>>) -> String2D-Liste in CSV-Text serialisieren
rowToStringfnc rowToString(row: List<String>) -> StringEinzelne Zeile serialisieren (Quote-Escaping)

Beispiel

import tinox.core.csv;
import tinox.core.fs;

fn main() -> Int64
{
    let content: String = Fs::readFile("data.csv");
    let rows: List<List<String>> = Csv::parse(content);

    // Erste Zeile = Header
    let header: List<String> = rows[0];
    println(header[0]); // z.B. "Name"

    // Daten schreiben
    let newData: List<List<String>> = [
        ["Name", "Alter", "Stadt"],
        ["Alice", "30", "Berlin"],
        ["Bob",   "25", "München"]
    ];
    let csv: String = Csv::toString(newData);
    Fs::writeFile("output.csv", csv);
    return 0;
}
