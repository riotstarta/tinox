graph

import tinox.core.graph;

Generische Graphdatenstruktur mit gewichteten Kanten, BFS, DFS und Dijkstra-Algorithmus.

Klassen

Beispiel

import tinox.core.graph;

fn main() -> Int64
{
    let g: Graph<String> = Graph::new();
    Graph::addNode(g, "A", "Start");
    Graph::addNode(g, "B", "Mitte");
    Graph::addNode(g, "C", "Ende");
    Graph::addEdge(g, "A", "B", 1.0);
    Graph::addEdge(g, "B", "C", 2.0);
    Graph::addEdge(g, "A", "C", 10.0);

    // BFS-Traversierung
    Graph::bfs(g, "A", fnc(id: String) -> Nothing { println(id); });

    // Kürzester Pfad A → C = 3.0
    let dist: Float64 = Graph::dijkstra(g, "A", "C");
    println(dist.toString()); // "3.0"
    return 0;
}
