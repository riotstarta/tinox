trie

import tinox.core.trie;

Trie-Datenstruktur für präfixbasierte Stringsuche und Autocomplete-Funktionalität.

Klassen

Trie

MethodeSignaturBeschreibung
newfn new() -> TrieLeeren Trie erstellen
insertfn insert(trie: Trie, word: String) -> NothingWort einfügen
searchfn search(trie: Trie, word: String) -> BoolExakte Wortsuche
startsWithfn startsWith(trie: Trie, prefix: String) -> BoolGibt es Wörter mit diesem Präfix?
autocompletefn autocomplete(trie: Trie, prefix: String) -> List<String>Alle Wörter mit gegebenem Präfix

Beispiel

import tinox.core.trie;

fn main() -> Int64
{
    let t: Trie = Trie::new();
    Trie::insert(t, "tinox");
    Trie::insert(t, "tiger");
    Trie::insert(t, "time");
    Trie::insert(t, "table");

    println(Trie::search(t, "tinox").toString()); // "true"
    println(Trie::search(t, "tin").toString());   // "false"
    println(Trie::startsWith(t, "ti").toString()); // "true"

    let matches: List<String> = Trie::autocomplete(t, "ti");
    // ["tinox", "tiger", "time"]
    return 0;
}
