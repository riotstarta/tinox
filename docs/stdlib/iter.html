iter

import tinox.core.iter;

Iterator-Utilities: range, repeat, cycle, enumerate, zip sowie map/filter/reduce für Listen.

Klassen

Iter

Beispiel

import tinox.core.iter;

fn main() -> Int64
{
    // range
    let it: Iterator = Iter::range(0, 3);
    while it.hasNext() { println(it.next().toString()); } // 0, 1, 2

    // map + filter
    let nums: List<Int64> = [1, 2, 3, 4, 5];
    let evens: List<Int64> = Iter::filter(nums, fnc(n: Int64) -> Bool { return n % 2 == 0; });
    let doubled: List<Int64> = Iter::map(evens, fnc(n: Int64) -> Int64 { return n * 2; });

    // reduce: Summe
    let sum: Int64 = Iter::reduce(doubled, 0, fnc(acc: Int64, x: Int64) -> Int64 { return acc + x; });
    println(sum.toString()); // "12"  (2*2 + 4*2 = 12)

    // enumerate
    let pairs = Iter::enumerate(["a", "b", "c"]);
    return 0;
}
