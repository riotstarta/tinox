crypto

import tinox.core.crypto;

Kryptografische Funktionen: Hashing (MD5, SHA-256), HMAC, AES-Verschlüsselung und Schlüsselableitung (PBKDF2).

Klassen

Crypto

MethodeSignaturBeschreibung
md5fnc md5(data: String) -> StringMD5-Hash als Hex-String
sha256fnc sha256(data: String) -> StringSHA-256-Hash als Hex-String
hmacSha256fnc hmacSha256(data, key: String) -> StringHMAC-SHA256-Signatur
aesEncryptfnc aesEncrypt(data, key: String) -> StringAES-Verschlüsselung
aesDecryptfnc aesDecrypt(data, key: String) -> StringAES-Entschlüsselung
generateKeyfnc generateKey(length: Int64) -> StringZufälligen Schlüssel generieren
pbkdf2fnc pbkdf2(password, salt: String, iterations, keyLength: Int64) -> StringPasswort-basierte Schlüsselableitung

Beispiel

import tinox.core.crypto;

fn main() -> Int64
{
    // Passwort hashen
    let hash: String = Crypto::sha256("meinPasswort123");
    println(hash);

    // PBKDF2 Schlüsselableitung (sicherer für Passwörter)
    let key: String = Crypto::pbkdf2("passwort", "salz123", 1000, 32);

    // AES Verschlüsselung
    let secret: String = Crypto::generateKey(32);
    let encrypted: String = Crypto::aesEncrypt("Geheime Nachricht", secret);
    let decrypted: String = Crypto::aesDecrypt(encrypted, secret);

    // HMAC für API-Signaturen
    let sig: String = Crypto::hmacSha256("payload", "apiSecret");
    return 0;
}
