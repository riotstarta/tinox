zip

import tinox.core.zip;

ZIP-Archiv-Handling: erstellen, öffnen, auflisten, extrahieren, hinzufügen und entfernen von Dateien.

Klassen

Zip

MethodeSignaturBeschreibung
createfnc create(path: String) -> ZipArchiveNeues leeres ZIP-Archiv erstellen
openfnc open(path: String) -> ZipArchiveBestehendes ZIP-Archiv öffnen
listEntriesfnc listEntries(path: String) -> List<ZipEntry>Alle Einträge im Archiv auflisten
extractFilefnc extractFile(archive: ZipArchive, entryName: String) -> StringDatei aus Archiv extrahieren
addFilefnc addFile(archive: ZipArchive, name: String, content: String) -> UnitDatei zum Archiv hinzufügen
removeFilefnc removeFile(archive: ZipArchive, entryName: String) -> UnitDatei aus Archiv entfernen
closefnc close(archive: ZipArchive) -> UnitArchiv schließen

ZipArchive / ZipEntry

FeldTypBeschreibung
archive.pathStringPfad zum ZIP-Archiv
archive.entriesList<ZipEntry>Liste aller Einträge
entry.nameStringDateiname im Archiv
entry.sizeInt64Dateigröße in Bytes

Beispiel

import tinox.core.zip;

fn main() -> Int64
{
    // Neues Archiv erstellen
    let archive: ZipArchive = Zip::create("output.zip");
    Zip::addFile(archive, "hello.txt", "Hallo Tinox!");
    Zip::addFile(archive, "data.json", "{\"key\":\"value\"}");
    Zip::close(archive);

    // Bestehendes öffnen und auflisten
    let existing: ZipArchive = Zip::open("output.zip");
    let i: Int64 = 0;
    while i < existing.entries.len() {
        println(existing.entries[i].name + " (" + existing.entries[i].size.toString() + " bytes)");
    }

    // Datei extrahieren
    let content: String = Zip::extractFile(existing, "hello.txt");
    println(content); // "Hallo Tinox!"
    return 0;
}
