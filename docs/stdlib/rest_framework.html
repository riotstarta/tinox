rest_framework

import tinox.core.rest_framework;

Annotations-basiertes REST-Framework für deklaratives API-Routing mit @GET, @POST, @PUT, @PATCH, @DELETE, @Path, @Produces, @Consumes, @StatusCode und @Auth. Inkl. Dependency-Injection-Annotationen und Middleware-Support.

Beispiel

import tinox.core.rest_framework;

@Path("/api/users")
class UserController : RestController
{
    @GET("/")
    @Produces("application/json")
    fnc list(ctx: HttpContext) -> Nothing
    {
        ctx.response.json("[{\"id\":1,\"name\":\"Alice\"}]");
    }

    @GET("/:id")
    @Produces("application/json")
    fnc getOne(ctx: HttpContext) -> Nothing
    {
        let id: String = ctx.request.getParam("id");
        ctx.response.json("{\"id\":" + id + "}");
    }

    @POST("/")
    @Consumes("application/json")
    @StatusCode(201)
    @Produces("application/json")
    fnc create(ctx: HttpContext) -> Nothing
    {
        let data: JsonValue = ctx.request.json();
        ctx.response.status(201).json("{\"created\":true}");
    }

    @DELETE("/:id")
    @Auth("bearer")
    fnc remove(ctx: HttpContext) -> Nothing
    {
        ctx.response.status(204).text("");
    }
}

fn main() -> Int64
{
    let app: RestApi = RestApi::new(8080);
    app.enableCors("*")
       .defaultJson()
       .register(new UserController());

    // Direkte Route ohne Controller
    app.get("/ping", ctx => {
        ctx.response.text("pong");
    });

    app.start(); // Blockierend
    return 0;
}
