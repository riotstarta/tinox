amqp091

Generated by tinox doc

AMQP-0-9-1 client (no broker) for message queue brokers such as RabbitMQ. Explicit publish/consume API, one channel per connection; amqps:// (TLS) is supported.

ModuleVersionGroup
socket1.0.0tinox.core

Publish consume

import tinox.core.amqp091;
import tinox.core.encoding;

class Main
{
    fnc main() -> Int32
    {
        let connection: AmqpConnection091 = AmqpConnection091::connect("localhost", 5672, "/", "guest", "guest");
        let amqpChannel: AmqpChannel091 = AmqpChannel091::open(connection);

        let queue: String = amqpChannel.declareQueue("jobs", true, false, false);
        let body: List<Int64> = Encoding::toUtf8("hello");
        amqpChannel.publish("", queue, body, "text/plain");

        amqpChannel.consume(queue);
        let msg: AmqpMessage091 = amqpChannel.nextMessage();
        println(msg.body.len().toString() + " bytes received");
        amqpChannel.ack(msg.deliveryTag);

        connection.close();
        return 0;
    }
}
