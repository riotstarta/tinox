websocket

Generated by tinox doc

RFC 6455 WebSocket server (WsServer/Ws) and client (WsClient), built on the HTTP server's raw connection-handle layer. v1 is an explicit accept/read-loop API (no lambda handler), serving one connection at a time; Ws::readMessage handles ping/pong/close automatically. TLS variants (wss://) are available via WsServer::listenTls/acceptTls.

Echo server

import tinox.core.websocket;

class Main
{
    fnc main() -> Int32
    {
        let srv: Int64 = WsServer::listen(8080);
        let conn: Int64 = WsServer::accept(srv);

        while true
        {
            let f: WsFrame = Ws::readMessage(conn);
            if f.opcode != 1
            {
                break;
            }
            Ws::sendText(conn, Ws::text(f));
        }

        Ws::close(conn);
        return 0;
    }
}

import tinox.core.websocket;

class Main
{
    fnc main() -> Int32
    {
        let conn: Int64 = WsClient::connect("localhost", 8080, "/");
        WsClient::sendText(conn, "hello from client");

        let f: WsFrame = WsClient::readMessage(conn);
        println(WsClient::text(f));

        WsClient::close(conn);
        return 0;
    }
}
