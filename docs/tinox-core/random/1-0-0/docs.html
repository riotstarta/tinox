random

Generated by tinox doc

Pseudo-random number and collection utilities: nextInt/nextFloat/nextBool, picking one random element (pickOne), shuffling or sampling a List<T> without replacement, and generating a random alphanumeric string.

Random values

import tinox.core.random;

class Main
{
    fnc main() -> Int32
    {
        println("dice roll: " + (Random::nextInt(6) + 1).toString());
        println("coin flip: " + Random::nextBool().toString());
        println("token: " + Random::generateString(12));

        let names: List<String> = ["Ada", "Grace", "Alan", "Barbara"];
        println("picked: " + Random::pickOne(names));

        let shuffled: List<String> = Random::shuffle(names);
        println("shuffled count: " + shuffled.len().toString());
        return 0;
    }
}
