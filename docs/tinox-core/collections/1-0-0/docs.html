collections

Generated by tinox doc

Generic data structures: Stack<T> (LIFO), Queue<T> (FIFO), and Pair<T, U> (a two-value tuple). The Collections class also provides newStack()/newQueue()/newPair() factory helpers.

Stack and queue

import tinox.core.collections;

class Main
{
    fnc main() -> Int32
    {
        let stack: Stack<Int64> = Stack::new();
        stack.push(1);
        stack.push(2);
        stack.push(3);
        println("stack pop: " + stack.pop().toString()); // 3 (LIFO)

        let queue: Queue<Int64> = Queue::new();
        queue.enqueue(1);
        queue.enqueue(2);
        queue.enqueue(3);
        println("queue dequeue: " + queue.dequeue().toString()); // 1 (FIFO)

        let pair: Pair<String, Int64> = Pair::new("age", 30);
        println(pair.getFirst() + " = " + pair.getSecond().toString());
        return 0;
    }
}
