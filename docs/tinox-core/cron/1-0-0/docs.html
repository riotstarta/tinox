cron

Generated by tinox doc

Standard 5-field cron expression parsing (Cron::parse/matches/nextRun) plus a CronScheduler that runs registered callbacks on a tick loop -- useful for periodic in-process jobs without an external scheduler.

Parse and match

import tinox.core.cron;
import tinox.core.time;

class Main
{
    fnc main() -> Int32
    {
        // every day at 09:00
        let expr: CronExpr = Cron::parse("0 9 * * *");

        let now: Int64 = Time::now();
        println("matches now: " + Cron::matches(expr, now).toString());
        println("next run at: " + Cron::nextRun(expr).toString());
        return 0;
    }
}
