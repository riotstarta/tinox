fs

Generated by tinox doc

Basic filesystem operations: read/write/append/copy/move/delete files, list and create/delete directories. moveFile is copy+delete (no atomic rename), and createDirectory is not recursive -- create parent directories yourself first.

Read write file

import tinox.core.fs;

class Main
{
    fnc main() -> Int32
    {
        Fs::writeFile("/tmp/greeting.txt", "hello from tinox\n");
        Fs::appendFile("/tmp/greeting.txt", "second line\n");

        let content: String = Fs::readFile("/tmp/greeting.txt");
        println(content);

        Fs::copyFile("/tmp/greeting.txt", "/tmp/greeting-copy.txt");

        let files: List<String> = Fs::listDirectory("/tmp");
        println("entries: " + files.len().toString());

        Fs::deleteFile("/tmp/greeting-copy.txt");

        return 0;
    }
}
