toml

Generated by tinox doc

A minimal TOML parser and serializer. Parses TOML text into a dynamically-typed TomlValue (tables, arrays, strings, ints, floats, bools) via Toml::parse, and renders a TomlValue back to TOML text via Toml::stringify. Note: nested [section] tables are recognized but key/value lines under a section are currently stored as flat top-level keys, not nested inside that section's table.

import tinox.core.toml;

class Main
{
    fnc main() -> Int32
    {
        let content: String = "name = \"tinox-central\"\nport = 8080\ndebug = true\n";
        let value: TomlValue = Toml::parse(content);
        let table: Map<String, TomlValue> = value.asTable();

        println(table["name"].asString());
        println(table["port"].asInt().toString());
        println(table["debug"].asBool().toString());

        return 0;
    }
}
