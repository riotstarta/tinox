mathf

Generated by tinox doc

Float64 math functions: sqrt, pow, trig (sin/cos/tan/atan2), log/log10/exp, floor/ceil/round, min/max/clamp/lerp. The floating-point counterpart to math's Int64-only helpers.

import tinox.core.mathf;

class Main

    {
        let root: Float64 = Mathf::sqrt(2.0);

        println("round(3.6) = " + Mathf::round(3.6).toString());
        println("clamp(15.0, 0.0, 10.0) = " + Mathf::clamp(15.0, 0.0, 10.0).toString());
        println("lerp(0.0, 100.0, 0.25) = " + Mathf::lerp(0.0, 100.0, 0.25).toString());
        return 0;
    }
}
