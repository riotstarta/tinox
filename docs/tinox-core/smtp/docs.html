smtp

Generated by tinox doc

SMTP client (RFC 5321) for sending plain-text email: plaintext connect + STARTTLS (port 587), implicit TLS (port 465), SASL PLAIN auth. Typical use: password-reset and signup-confirmation emails.

import tinox.core.smtp;

class Main
{
    fnc main() -> Int32
    {
        let client: SmtpClient = SmtpClient::connect("smtp.example.org", 587);
        if client.conn <= 0
        {
            println("connect failed: " + client.errorMessage);
            return 1;
        }

        if !client.startTls()
        {
            println("STARTTLS failed");
            return 1;
        }

        if !client.authPlain("user", "pass")
        {
            println("auth failed");
            return 1;
        }

        client.send("me@example.org", ["you@example.org"], "Hi", "Hello from Tinox!");
        client.close();
        return 0;
    }
}
