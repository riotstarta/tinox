json

Generated by tinox doc

JSON parsing and generation. Json::serialize/deserialize<T> work directly with @JsonSerializable-annotated classes; Json::parse/stringify and the JsonValue wrapper (getString/getInt/asTable/asList/isObject/...) give untyped access for dynamic/unknown-shape JSON.

Parse and read

import tinox.core.json;

class Main
{
    fnc main() -> Int32
    {
        let value: JsonValue = Json::parse("{\"name\": \"tinox\", \"stable\": true}");
        println(value.getField("name").getString());
        println(value.getField("stable").getBool().toString());

        let map: Map<String, JsonValue> = value.asTable();
        println(map.keys().len().toString());

        println(Json::stringify(value));
        return 0;
    }
}
