logger

Generated by tinox doc

Leveled logging (debug/info/warn/error) with a per-logger name and minimum level filter. Plain-text by default; set TINOX_LOG_JSON=true to switch every Logger instance to structured JSON output instead, no code change needed.

Leveled logging

import tinox.core.logger;

class Main
{
    fnc main() -> Int32
    {
        let log: Logger = Logger::new("Main");
        Logger::setLevel(log, LogLevel::Warn);

        Logger::debug(log, "this is filtered out");
        Logger::info(log, "this is filtered out too");
        Logger::warn(log, "disk usage above 80%");
        Logger::error(log, "connection refused");
        return 0;
    }
}
